### 2.1 Calculate delta distance and step distance.
```
// Calculate delta distance and step distance.

// direction vectors of angle
ray_dir_x := math.cos(angle)
ray_dir_y := math.sin(angle)

// Distance the ray has to travel to go from 1 vertical/horizontal grid line to the next
// prevent division-by-zero if running parallel to an axis
delta_dist_x := (ray_dir_x == 0) ? 1e30 : math.abs(1.0 / ray_dir_x)
delta_dist_y := (ray_dir_y == 0) ? 1e30 : math.abs(1.0 / ray_dir_y)

// lenght of ray from current position to 1st grid line
side_dist_x: f32 = 0.0
side_dist_y: f32 = 0.0

// direction to step in the grid map (-1[back] or +1[front]). need to do because array is flat.
step_x: i32 = 0
step_y: i32 = 0

// current integer square of the map the player is in
map_x := i32(PX)
map_y := i32(PY)

// Calculate step and initial side_dist
if ray_dir_x < 0 {
    step_x = -1
    side_dist_x = (PX - f32(map_x)) * delta_dist_x
} else {
    step_x = 1
    side_dist_x = (f32(map_x) + 1.0 - PX) * delta_dist_x
}

if ray_dir_y < 0 {
    step_y = -1
    side_dist_y = (PY - f32(map_y)) * delta_dist_y
} else {
    step_y = 1
    side_dist_y = (f32(map_y) + 1.0 - PY) * delta_dist_y
}
```

### 2.2 The DDA Jump Loop
```
// The DDA Jump Loop
// side_hit: 0 for vertical wall (X-axis grid line), 1 for horizontal wall (Y-axis grid line)
side_hit: i32 = 0 
for MAP[map_y * MAP_W + map_x] == 0 {
    // jump to the next closest grid line (either X line or Y line)
    if side_dist_x < side_dist_y {
        side_dist_x += delta_dist_x
        map_x += step_x
        side_hit = 0
    } else {
        side_dist_y += delta_dist_y
        map_y += step_y
        side_hit = 1
    }
}
```

### 2.3 Draw the walls
```
// draw the walls

// perfect perpendicular distance to the wall (automatically fixes fish-eye!)
perp_wall_dist: f32 = 0.0
if side_hit == 0 {
    perp_wall_dist = side_dist_x - delta_dist_x
} else {
    perp_wall_dist = side_dist_y - delta_dist_y
}
if perp_wall_dist < 0.01 do perp_wall_dist = 0.01

h := i32(f32(HEIGHT) / perp_wall_dist)
rl.DrawLine(x, HALF_HEIGHT - h/2, x, HALF_HEIGHT + h/2, rl.RED)
```
